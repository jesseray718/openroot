def sha256_hash(content: str) -> str:
"""Generate SHA-256 hash of content."""
return hashlib.sha256(content.encode('utf-8')).hexdigest()

def generate_id(doc_type: str) -> str:
"""Generate unique document ID."""
ts = datetime.utcnow().strftime("%Y%m%d%H%M%S")
return f"{doc_type.upper()}_{ts}"

def front_matter(doc_data: Dict[str, Any]) -> str:
"""Generate YAML front matter for document."""
return yaml.dump({
"id": doc_data.get("id", generate_id(doc_data.get("type", "doc"))),
"timestamp": doc_data.get("timestamp", CONFIG["timestamp_iso"]),
"type": doc_data.get("type", "document"),
"parent": doc_data.get("parent", None),
"hash": None, # Will be calculated after content
"status": doc_data.get("status", "active"),
"agape_score": doc_data.get("agape_score", None)
}, sort_keys=False).strip()

def write_document(base_path: str, filename: str, template_func) -> str:
"""Write a document from template and return its hash."""
template_data = template_func()
content = template_data["content"]
# Calculate hash
doc_hash = sha256_hash(content)

# Create complete document with updated hash
full_content = content.replace("hash: null", f"hash: {doc_hash}")

# Write file
filepath = os.path.join(base_path, filename)
with open(filepath, 'w', encoding='utf-8') as f:
f.write(full_content)

print(f"[✓] Wrote: {filepath} (hash: {doc_hash[:16]}...)")
return doc_hash

def sha256_hash(content: str) -> str:
    """Generate SHA-256 hash of content."""
    return hashlib.sha256(content.encode('utf-8')).hexdigest()

def generate_id(doc_type: str) -> str:
    """Generate unique document ID."""
    ts = datetime.utcnow().strftime("%Y%m%d%H%M%S")
    return f"{doc_type.upper()}_{ts}"

        "status": doc_data.get("status", "active"),
        "agape_score": doc_data.get("agape_score", None)
    }
    # Ensure consistent ordering
    return yaml.dump(fm, sort_keys=False).strip()

def build_merkle_tree(base_path: str, output_file: str):
    """Build a Merkle tree from all generated documents."""
    print("\n[PHASE 3] Building Merkle Tree...")
    
    # Collect all document hashes
    file_hashes = {}
    
    # Define paths to hash
    patterns = [
        "tokens/*.md",
        "THESIS.md",
        "v1-issue-body.md",
        "aero-disc/docs/*.md",
        "tokens/WORK_SPECIFICATION.md"
    ]
    
    for pattern in patterns:
        # Simple glob-like expansion
        if '*' in pattern:
            dir_part, file_part = pattern.rsplit('/', 1)
            dir_path = os.path.join(base_path, dir_part)
            if os.path.exists(dir_path):
                for f in os.listdir(dir_path):
                    if f.endswith('.md'):
                        full_path = os.path.join(dir_path, f)
                        rel_path = os.path.relpath(full_path, base_path)
                        try:
                            with open(full_path, 'r', encoding='utf-8') as file:
                                content = file.read()
                                h = sha256_hash(content)
                                file_hashes[rel_path] = h
                                print(f"  Hashed: {rel_path} -> {h[:16]}...")
                        except Exception as e:
                            print(f"  [!] Error hashing {rel_path}: {e}")
        else:
            full_path = os.path.join(base_path, pattern)
            if os.path.exists(full_path):
                rel_path = pattern
                try:
                    with open(full_path, 'r', encoding='utf-8') as file:
                        content = file.read()
                        h = sha256_hash(content)
                        file_hashes[rel_path] = h
                        print(f"  Hashed: {rel_path} -> {h[:16]}...")
                except Exception as e:
                    print(f"  [!] Error hashing {rel_path}: {e}")
    
    if not file_hashes:
        print("[!] No files found to hash.")
        return None

    # Build Merkle tree using Python for proper pairing
    keys = sorted(file_hashes.keys())
    
    # Pad to power of 2
    while len(keys) & (len(keys) - 1):
        keys.append(keys[-1])  # Duplicate last for padding

    # Build tree bottom-up
    tree_levels = []
    current_level = [(k, file_hashes[k]) for k in keys]
    tree_levels.append(current_level)

    while len(current_level) > 1:
        next_level = []
        for i in range(0, len(current_level), 2):
            left_key, left_hash = current_level[i]
            right_key, right_hash = current_level[i+1] if i+1 < len(current_level) else (left_key, left_hash)
            
            combined = left_hash + right_hash
            parent_hash = hashlib.sha256(combined.encode()).hexdigest()
            next_level.append((f"{left_key}+{right_key}", parent_hash))
        
        tree_levels.append(next_level)
        current_level = next_level

    merkle_root = current_level[0][1]
    root_idx = len(tree_levels) - 1

    output = {
        "merkle_root": merkle_root,
        "root_depth": root_idx,
        "document_count": len(keys),
        "documents": {k: v for k, v in file_hashes.items()},
        "tree_levels": len(tree_levels),
        "timestamp": CONFIG["timestamp_iso"]
    }

    output_path = os.path.join(base_path, output_file)
    with open(output_path, 'w', encoding='utf-8') as f:
        json.dump(output, f, indent=2)
    
    print(f"[✓] Merkle tree saved to: {output_path}")
    print(f"    Root: {merkle_root[:32]}...")
    return output_path

def sha256_hash(content: str) -> str:

def normalize_path(path: str) -> str:
    """Normalize path for consistent display."""
    # Convert absolute paths to relative labels
    for scan_path in CONFIG["scan_paths"]:

def extract_yaml_frontmatter(content: str) -> tuple:
    """Separate YAML frontmatter from body content."""
    if content.startswith("---"):
        end_marker = content.find("---", 3)
        if end_marker != -1:
            fm_text = content[4:end_marker].strip()
            body = content[end_marker + 3:].strip()
            try:
                fm = yaml.safe_load(fm_text)
                return fm, body
            except:
                return {}, content
    return {}, content

def collect_syncthing_folders(paths: List[str]) -> List[Dict[str, Any]]:
    """Collect all folders from Syncthing directories."""
    all_entries = []
    
    for base_path in paths:
        if not os.path.exists(base_path):
            print(f"[!] Skipping non-existent path: {base_path}")
            continue
        
        for root, dirs, files in os.walk(base_path):
            for dir_name in dirs:
                dir_path = os.path.join(root, dir_name)
                relative = os.path.relpath(dir_path, base_path)
                
                # Count files in folder
                file_count = sum(len(files) for r, d, files in os.walk(dir_path))
                size_bytes = sum(os.path.getsize(os.path.join(r,f)) 
                               for r, d, files in os.walk(dir_path) 
                               for f in files if os.path.isfile(os.path.join(r,f)))
                
                entry = {
                    "type": "folder",
                    "original_path": dir_path,
                    "relative_path": relative,
                    "name": dir_name,
                    "parent": os.path.basename(root),
                    "file_count": file_count,
                    "estimated_size_mb": round(size_bytes / (1024*1024), 2),
                    "hash_preview": sha256_hash(dir_path)[:12]
                }
                all_entries.append(entry)
            
            # Also add large files directly
            for file_name in files:
                file_path = os.path.join(root, file_name)
                if os.path.getsize(file_path) > 10000:  # Skip tiny files
                    relative = os.path.relpath(file_path, base_path)
                    
                    try:
                        with open(file_path, 'r', encoding='utf-8') as f:
                            content = f.read(1000)
                        content_hash = sha256_hash(content)
                    except:
                        content_hash = "binary_or_unreadable"
                    
                    entry = {
                        "type": "file",
                        "original_path": file_path,
                        "relative_path": relative,
                        "name": file_name,
                        "parent": os.path.basename(root),
                        "size_bytes": os.path.getsize(file_path),
                        "hash_preview": content_hash[:12],
                        "extension": os.path.splitext(file_name)[1].lower()
                    }
                    all_entries.append(entry)
    
    # Sort by path for consistent output
    all_entries.sort(key=lambda x: x["relative_path"])
    return all_entries

build_merkle_tree() {
log_header "Building Merkle Tree"

mkdir -p "$AGAPENET_DATA"

# Collect all document hashes
declare -A file_hashes

for doc in "${BASE_PATH}"/tokens/*.md \
"${BASE_PATH}"/THESIS.md \
"${BASE_PATH}"/v1-issue-body.md \
"${BASE_PATH}"/aero-disc/docs/*.md; do

rel_path="${doc#$BASE_PATH/}"
file_hashes["$rel_path"]="$hash"
log_info "Hashed: $rel_path → ${hash:0:16}..."
fi
done

# Build Merkle tree using Python for proper pairing
python3 << PYTHON_MERKLE
import json
import hashlib
from pathlib import Path

file_hashes = $(declare -p file_hashes)

# Pad to power of 2
keys = sorted(file_hashes.keys())
while len(keys) & (len(keys) - 1):
keys.append(keys[-1]) # Duplicate last for padding

# Build tree bottom-up
tree_levels = []
current_level = [(k, file_hashes[k]) for k in keys]
tree_levels.append(current_level)

while len(current_level) > 1:
next_level = []
for i in range(0, len(current_level), 2):
left_key, left_hash = current_level[i]
right_key, right_hash = current_level[i+1] if i+1 < len(current_level) else (left_key, left_hash)

combined = left_hash + right_hash
parent_hash = hashlib.sha256(combined.encode()).hexdigest()
next_level.append((f"{left_key}+{right_key}", parent_hash))

tree_levels.append(next_level)
current_level = next_level

merkle_root = current_level[0][1]
root_idx = len(tree_levels) - 1

output = {
"merkle_root": merkle_root,
"root_depth": root_idx,
"document_count": len(keys),
"documents": {k: v for k, v in file_hashes.items()},
"tree_levels": len(tree_levels)
}

Path("${MERKLE_ROOT_FILE}").write_text(json.dumps(output, indent=2))
print(f"Merkle root: {merkle_root}")
PYTHON_MERKLE

log_info "Merkle tree saved to: $MERKLE_ROOT_FILE"
}

        "tokens/WORK_SPECIFICATION.md"
    ]
    
    for pattern in patterns:
        # Simple glob-like expansion
        if '*' in pattern:
            dir_part, file_part = pattern.rsplit('/', 1)
            dir_path = os.path.join(base_path, dir_part)
            if os.path.exists(dir_path):
                for f in os.listdir(dir_path):
                    if f.endswith('.md'):
                        full_path = os.path.join(dir_path, f)
                        rel_path = os.path.relpath(full_path, base_path)
                        try:
                            with open(full_path, 'r', encoding='utf-8') as file:
                                content = file.read()
                                h = sha256_hash(content)
                                file_hashes[rel_path] = h
                                print(f"  Hashed: {rel_path} -> {h[:16]}...")
                        except Exception as e:
                            print(f"  [!] Error hashing {rel_path}: {e}")
        else:
            full_path = os.path.join(base_path, pattern)
            if os.path.exists(full_path):
                rel_path = pattern
                try:
                    with open(full_path, 'r', encoding='utf-8') as file:
                        content = file.read()
                        h = sha256_hash(content)
                        file_hashes[rel_path] = h
                        print(f"  Hashed: {rel_path} -> {h[:16]}...")
                except Exception as e:
                    print(f"  [!] Error hashing {rel_path}: {e}")
    
    if not file_hashes:
        print("[!] No files found to hash.")
        return None

    # Build Merkle tree using Python for proper pairing
    keys = sorted(file_hashes.keys())
    
    # Pad to power of 2
    while len(keys) & (len(keys) - 1):
        keys.append(keys[-1])  # Duplicate last for padding

    # Build tree bottom-up
    tree_levels = []
    current_level = [(k, file_hashes[k]) for k in keys]
    tree_levels.append(current_level)

    while len(current_level) > 1:
        next_level = []
        for i in range(0, len(current_level), 2):
            left_key, left_hash = current_level[i]
            right_key, right_hash = current_level[i+1] if i+1 < len(current_level) else (left_key, left_hash)
            
            combined = left_hash + right_hash
            parent_hash = hashlib.sha256(combined.encode()).hexdigest()
            next_level.append((f"{left_key}+{right_key}", parent_hash))
        
        tree_levels.append(next_level)
        current_level = next_level

    merkle_root = current_level[0][1]
    root_idx = len(tree_levels) - 1

    output = {
        "merkle_root": merkle_root,
        "root_depth": root_idx,
        "document_count": len(keys),
        "documents": {k: v for k, v in file_hashes.items()},
        "tree_levels": len(tree_levels),
        "timestamp": CONFIG["timestamp_iso"]
    }

    output_path = os.path.join(base_path, output_file)
    with open(output_path, 'w', encoding='utf-8') as f:
        json.dump(output, f, indent=2)
    
    print(f"[✓] Merkle tree saved to: {output_path}")
    print(f"    Root: {merkle_root[:32]}...")
    return output_path

    """Separate YAML frontmatter from body content."""

            fm_text = content[4:end_marker].strip()
            body = content[end_marker + 3:].strip()

def collect_syncthing_folders(paths: List[str]) -> List[Dict[str, Any]]:
    """Collect all folders from Syncthing directories."""
    all_entries = []
    
    for base_path in paths:
        if not os.path.exists(base_path):
            print(f"[!] Skipping non-existent path: {base_path}")
            continue
        
        for root, dirs, files in os.walk(base_path):
            for dir_name in dirs:
                dir_path = os.path.join(root, dir_name)
                relative = os.path.relpath(dir_path, base_path)
                
                # Count files in folder
                file_count = sum(len(files) for r, d, files in os.walk(dir_path))
                size_bytes = sum(os.path.getsize(os.path.join(r,f)) 
                               for r, d, files in os.walk(dir_path) 
                               for f in files if os.path.isfile(os.path.join(r,f)))
                
                entry = {
                    "type": "folder",
                    "original_path": dir_path,
                    "relative_path": relative,
                    "name": dir_name,
                    "parent": os.path.basename(root),
                    "file_count": file_count,
                    "estimated_size_mb": round(size_bytes / (1024*1024), 2),
                    "hash_preview": sha256_hash(dir_path)[:12]
                }
                all_entries.append(entry)
            
            # Also add large files directly
            for file_name in files:
                file_path = os.path.join(root, file_name)
                if os.path.getsize(file_path) > 10000:  # Skip tiny files
                    relative = os.path.relpath(file_path, base_path)
                    
                    try:
                        with open(file_path, 'r', encoding='utf-8') as f:
                            content = f.read(1000)
                        content_hash = sha256_hash(content)
                    except:
                        content_hash = "binary_or_unreadable"
                    
                    entry = {
                        "type": "file",
                        "original_path": file_path,
                        "relative_path": relative,
                        "name": file_name,
                        "parent": os.path.basename(root),
                        "size_bytes": os.path.getsize(file_path),
                        "hash_preview": content_hash[:12],
                        "extension": os.path.splitext(file_name)[1].lower()
                    }
                    all_entries.append(entry)
    
    # Sort by path for consistent output
    all_entries.sort(key=lambda x: x["relative_path"])
    return all_entries

build_merkle_tree() {
log_header "Building Merkle Tree"

mkdir -p "$AGAPENET_DATA"

# Collect all document hashes
declare -A file_hashes

for doc in "${BASE_PATH}"/tokens/*.md \
"${BASE_PATH}"/THESIS.md \
"${BASE_PATH}"/v1-issue-body.md \
"${BASE_PATH}"/aero-disc/docs/*.md; do

hash=$(compute_file_hash "$doc")
rel_path="${doc#$BASE_PATH/}"
file_hashes["$rel_path"]="$hash"
log_info "Hashed: $rel_path → ${hash:0:16}..."
fi
done

# Build Merkle tree using Python for proper pairing
python3 << PYTHON_MERKLE
import json
import hashlib
from pathlib import Path

file_hashes = $(declare -p file_hashes)

# Pad to power of 2
keys = sorted(file_hashes.keys())
while len(keys) & (len(keys) - 1):
keys.append(keys[-1]) # Duplicate last for padding

# Build tree bottom-up
tree_levels = []
current_level = [(k, file_hashes[k]) for k in keys]
tree_levels.append(current_level)

while len(current_level) > 1:
next_level = []
for i in range(0, len(current_level), 2):
left_key, left_hash = current_level[i]
right_key, right_hash = current_level[i+1] if i+1 < len(current_level) else (left_key, left_hash)

combined = left_hash + right_hash
parent_hash = hashlib.sha256(combined.encode()).hexdigest()
next_level.append((f"{left_key}+{right_key}", parent_hash))

tree_levels.append(next_level)
current_level = next_level

merkle_root = current_level[0][1]
root_idx = len(tree_levels) - 1

output = {
"merkle_root": merkle_root,
"root_depth": root_idx,
"document_count": len(keys),
"documents": {k: v for k, v in file_hashes.items()},
"tree_levels": len(tree_levels)
}

Path("${MERKLE_ROOT_FILE}").write_text(json.dumps(output, indent=2))
print(f"Merkle root: {merkle_root}")
PYTHON_MERKLE

log_info "Merkle tree saved to: $MERKLE_ROOT_FILE"
}

def build_merkle_tree(base_path: str, output_file: str):
    """Build a Merkle tree from all generated documents."""
    print("\n[PHASE 3] Building Merkle Tree...")
    
    # Collect all document hashes
    file_hashes = {}
    
    # Define paths to hash
    patterns = [
        "tokens/*.md",
        "THESIS.md",
        "v1-issue-body.md",
        "aero-disc/docs/*.md",
        "tokens/WORK_SPECIFICATION.md"
    ]
    
    for pattern in patterns:
        # Simple glob-like expansion
        if '*' in pattern:
            dir_part, file_part = pattern.rsplit('/', 1)
            dir_path = os.path.join(base_path, dir_part)
            if os.path.exists(dir_path):
                for f in os.listdir(dir_path):
                    if f.endswith('.md'):
                        full_path = os.path.join(dir_path, f)
                        rel_path = os.path.relpath(full_path, base_path)
                        try:
                            with open(full_path, 'r', encoding='utf-8') as file:
                                content = file.read()
                                h = sha256_hash(content)
                                file_hashes[rel_path] = h
                                print(f"  Hashed: {rel_path} -> {h[:16]}...")
                        except Exception as e:
                            print(f"  [!] Error hashing {rel_path}: {e}")
        else:
            full_path = os.path.join(base_path, pattern)
            if os.path.exists(full_path):
                rel_path = pattern
                try:
                    with open(full_path, 'r', encoding='utf-8') as file:
                        content = file.read()
                        h = sha256_hash(content)
                        file_hashes[rel_path] = h
                        print(f"  Hashed: {rel_path} -> {h[:16]}...")
                except Exception as e:
                    print(f"  [!] Error hashing {rel_path}: {e}")
    
    if not file_hashes:
        print("[!] No files found to hash.")
        return None

    # Build Merkle tree using Python for proper pairing
    keys = sorted(file_hashes.keys())
    
    # Pad to power of 2
    while len(keys) & (len(keys) - 1):
        keys.append(keys[-1])  # Duplicate last for padding

    # Build tree bottom-up
    tree_levels = []
    current_level = [(k, file_hashes[k]) for k in keys]
    tree_levels.append(current_level)

    while len(current_level) > 1:
        next_level = []
        for i in range(0, len(current_level), 2):
            left_key, left_hash = current_level[i]
            right_key, right_hash = current_level[i+1] if i+1 < len(current_level) else (left_key, left_hash)
            
            combined = left_hash + right_hash
            parent_hash = hashlib.sha256(combined.encode()).hexdigest()
            next_level.append((f"{left_key}+{right_key}", parent_hash))
        
        tree_levels.append(next_level)
        current_level = next_level

    merkle_root = current_level[0][1]
    root_idx = len(tree_levels) - 1

    output = {
        "merkle_root": merkle_root,
        "root_depth": root_idx,
        "document_count": len(keys),
        "documents": {k: v for k, v in file_hashes.items()},
        "tree_levels": len(tree_levels),
        "timestamp": CONFIG["timestamp_iso"]
    }

    output_path = os.path.join(base_path, output_file)
    with open(output_path, 'w', encoding='utf-8') as f:
        json.dump(output, f, indent=2)
    
    print(f"[✓] Merkle tree saved to: {output_path}")
    print(f"    Root: {merkle_root[:32]}...")
    return output_path

                all_entries.append(entry)
            
            # Also add large files directly
            for file_name in files:
                file_path = os.path.join(root, file_name)
                if os.path.getsize(file_path) > 10000:  # Skip tiny files
                    relative = os.path.relpath(file_path, base_path)
                    
                    try:
                        with open(file_path, 'r', encoding='utf-8') as f:
                            content = f.read(1000)
                        content_hash = sha256_hash(content)
                    except:
                        content_hash = "binary_or_unreadable"
                    
                    entry = {
                        "type": "file",
                        "original_path": file_path,
                        "relative_path": relative,
                        "name": file_name,
                        "parent": os.path.basename(root),
                        "size_bytes": os.path.getsize(file_path),
                        "hash_preview": content_hash[:12],
                        "extension": os.path.splitext(file_name)[1].lower()
                    }
                    all_entries.append(entry)
    
    # Sort by path for consistent output
    all_entries.sort(key=lambda x: x["relative_path"])
    return all_entries

def dynamic_price_calc(request: Dict[str, Any]) -> float:
    """AI-adjusted pricing based on demand, cash reserves, booking schedule."""
    
    base_price = {
        "desktop_pc": 5.0,
        "laptop": 8.0,
        "monitor": 3.0,
        "tv_small": 10.0,
        "tv_large": 20.0,
        "server_rack": 50.0,
        "appliances": 15.0
    }
    
    equipment_type = request.get("equipment_type", "desktop_pc")
    weight_multiplier = request.get("weight", 20) / 20
    
    # Demand factor (higher booking = higher price)
    booking_fullness = get_booking_percentage()
    demand_factor = 1.0 + (booking_fullness * 0.5)
    
    # Cash reserve factor (low reserves = lower price to attract work)
    cash_reserve = get_cash_reserves()
    urgency_discount = 0.8 if cash_reserve < 2000 else 1.0
    
    price = base_price.get(equipment_type, 10.0) * weight_multiplier * demand_factor * urgency_discount
    
    return round(price, 2)
